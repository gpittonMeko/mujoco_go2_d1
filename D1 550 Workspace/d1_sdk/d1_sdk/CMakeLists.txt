project(marm_code)
cmake_minimum_required(VERSION 3.5)

SET(CMAKE_CXX_STANDARD 17)

if(EXISTS "/usr/include/ddscxx")
  include_directories(/usr/include/ddscxx)
else()
  include_directories(/usr/local/include/ddscxx)
endif()
if(EXISTS "/usr/include/iceoryx/v2.0.2")
  include_directories(/usr/include/iceoryx/v2.0.2)
elseif(EXISTS "/usr/local/include/iceoryx/v2.0.2")
  include_directories(/usr/local/include/iceoryx/v2.0.2)
endif()
if(EXISTS "/usr/lib/aarch64-linux-gnu/libddsc.so" AND EXISTS "/usr/lib/aarch64-linux-gnu/libddscxx.so")
  set(DDSC_LIB /usr/lib/aarch64-linux-gnu/libddsc.so)
  set(DDSCXX_LIB /usr/lib/aarch64-linux-gnu/libddscxx.so)
else()
  set(DDSC_LIB ddsc)
  set(DDSCXX_LIB ddscxx)
endif()

if(EXISTS "/usr/local/lib/libunitree_go2_sdk.a" OR EXISTS "/usr/local/lib/libunitree_go2_sdk.so")
  link_libraries(unitree_go2_sdk unitree_go2_idl_cpp unitree_ros2_idl_cpp ${DDSC_LIB} ${DDSCXX_LIB} rt pthread)
else()
  link_libraries(unitree_sdk2 ${DDSC_LIB} ${DDSCXX_LIB} rt pthread)
endif()

add_executable(joint_angle_control src/joint_angle_control.cpp src/msg/ArmString_.cpp)

add_executable(d1_feedback_snapshot src/d1_feedback_snapshot.cpp src/msg/ArmString_.cpp src/msg/PubServoInfo_.cpp)
